# 2. Overwrite the Apache vhost config with the correct rules + flushpackets=on
cat > /usr/local/directadmin/data/users/admin/domains/aviator.betcheza.co.ke.conf << 'EOF'
# CrashBet API proxy — survives DirectAdmin config rebuilds
ProxyRequests Off
ProxyPreserveHost On

# SSE stream — MUST come first, flushpackets=on prevents Apache buffering
ProxyPass /api/game/stream http://127.0.0.1:3003/api/game/stream flushpackets=on
ProxyPassReverse /api/game/stream http://127.0.0.1:3003/api/game/stream

# General API
ProxyPass /api http://127.0.0.1:3003/api
ProxyPassReverse /api http://127.0.0.1:3003/api
EOF

# 3. Rebuild DirectAdmin's Apache config and reload
/usr/local/directadmin/directadmin rebuild-httpd 2>/dev/null || \
  /usr/local/directadmin/directadmin d rebuild_httpd 2>/dev/null || \
  systemctl reload httpd
